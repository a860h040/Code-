/********************************************************************** * STANDALONE 4-SECTION APIXABAN DURATION PIE CHART * * SOURCE: * Sheet1 * * OUTPUT: * MUE 4-Section Pie * * FOUR PIE SECTIONS: * * 1. Total Hip – Appropriate Duration * 2. Total Hip – Inappropriate Duration * 3. Total Knee – Appropriate Duration * 4. Total Knee – Inappropriate Duration * * STANDARD: * Knee = 12 days * Hip = 35 days * * IMPORTANT: * ------------------------------------------------------------ * - Does NOT modify Sheet1. * - Does NOT modify your other graph sheets. * - Does NOT depend on any other Apps Script functions. * - All counts and percentages are Google Sheets formulas. * - The four pie sections equal 100%. * - The legend contains patient count (n). * - The slices display percentage. * * RUN: * buildFourSectionDurationPieChart **********************************************************************/ function buildFourSectionDurationPieChart() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const SOURCE_SHEET = "Sheet1"; const OUTPUT_SHEET = "MUE 4-Section Pie"; const KNEE_DURATION = 12; const HIP_DURATION = 35; /******************************************************************** * GET SOURCE SHEET ********************************************************************/ const source = ss.getSheetByName( SOURCE_SHEET ); if (!source) { SpreadsheetApp .getUi() .alert( 'Could not find the sheet "' + SOURCE_SHEET + '".' ); return; } const lastRow = source.getLastRow(); const lastColumn = source.getLastColumn(); if (lastRow < 2) { SpreadsheetApp .getUi() .alert( "Sheet1 does not contain patient data." ); return; } /******************************************************************** * READ HEADERS ********************************************************************/ const headers = source .getRange( 1, 1, 1, lastColumn ) .getDisplayValues()[0]; /******************************************************************** * FIND SURGERY COLUMN ********************************************************************/ const surgeryColumn = findFourPieHeader_( headers, [ "Type of surgery", "Surgery Type", "Procedure Type", "Procedure" ] ); /******************************************************************** * FIND DURATION COLUMN ********************************************************************/ const durationColumn = findFourPieHeader_( headers, [ "Number of outpatient days supplied (total treatment duration)", "Total treatment duration", "Total Treatment Duration", "Total treatment duration (days)", "Treatment duration", "Total Duration", "Duration" ] ); if (!surgeryColumn) { SpreadsheetApp .getUi() .alert( 'Could not find the "Type of surgery" column.' ); return; } if (!durationColumn) { SpreadsheetApp .getUi() .alert( "Could not find the treatment-duration column." ); return; } /******************************************************************** * CONVERT COLUMN NUMBERS TO LETTERS ********************************************************************/ const surgeryLetter = fourPieColumnLetter_( surgeryColumn ); const durationLetter = fourPieColumnLetter_( durationColumn ); const safeSource = "'" + source .getName() .replace( /'/g, "''" ) + "'"; /******************************************************************** * CREATE / RESET OUTPUT SHEET ********************************************************************/ let output = ss.getSheetByName( OUTPUT_SHEET ); if (!output) { output = ss.insertSheet( OUTPUT_SHEET ); } else { /****************************************************************** * REMOVE OLD CHARTS ******************************************************************/ output .getCharts() .forEach( function(chart) { output.removeChart( chart ); } ); /****************************************************************** * BREAK MERGED CELLS ******************************************************************/ output .getDataRange() .getMergedRanges() .forEach( function(range) { range.breakApart(); } ); output.clear(); } /******************************************************************** * TITLE ********************************************************************/ output .getRange("A1:H1") .merge() .setValue( "Apixaban Duration Appropriateness by Surgery Type" ) .setFontSize(16) .setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#D9EAF7"); output .getRange("A2:H2") .merge() .setValue( "Four-section pie chart of total hip and total knee arthroplasty " + "patients based on duration appropriateness." ) .setWrap(true); output .getRange("A3:H3") .merge() .setValue( "Appropriate duration: knee = 12 days; hip = 35 days. " + "Any other documented duration is classified as inappropriate." ) .setWrap(true) .setFontStyle("italic"); /******************************************************************** * * PATIENT-LEVEL CALCULATION TABLE * * A = source row * B = procedure * C = duration * D = duration status * ********************************************************************/ output .getRange("A5:D5") .setValues([ [ "Source Row", "Procedure", "Duration (days)", "Duration Assessment" ] ]) .setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#EAF2F8"); const formulas = []; for ( let sourceRow = 2; sourceRow <= lastRow; sourceRow++ ) { const outputRow = sourceRow + 4; const surgeryRef = `${safeSource}!${surgeryLetter}${sourceRow}`; const durationRef = `${safeSource}!${durationLetter}${sourceRow}`; /****************************************************************** * PROCEDURE CLASSIFICATION * * Strictly identifies total hip / knee arthroplasty. ******************************************************************/ const procedureFormula = `=IF(` + `OR(` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total joint.*hip"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total hip"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"hip arthroplasty"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"(^|[^a-z])tha([^a-z]|$)")` + `),` + `"Hip",` + `IF(` + `OR(` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total joint.*knee"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total knee"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"knee arthroplasty"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"(^|[^a-z])tka([^a-z]|$)")` + `),` + `"Knee",` + `"Not included"` + `)` + `)`; /****************************************************************** * DURATION ******************************************************************/ const durationFormula = `=IFERROR(` + `VALUE(` + `REGEXEXTRACT(` + `TO_TEXT(${durationRef}),` + `"[0-9]+\\.?[0-9]*"` + `)` + `),` + `""` + `)`; /****************************************************************** * DURATION APPROPRIATENESS * * For exactly 4 pie sections: * * Appropriate = exact recommended duration * * Inappropriate = any other value, including missing duration ******************************************************************/ const assessmentFormula = `=IF(` + `B${outputRow}="Not included",` + `"Not included",` + `IF(` + `B${outputRow}="Hip",` + `IF(` + `C${outputRow}=${HIP_DURATION},` + `"Appropriate",` + `"Inappropriate"` + `),` + `IF(` + `B${outputRow}="Knee",` + `IF(` + `C${outputRow}=${KNEE_DURATION},` + `"Appropriate",` + `"Inappropriate"` + `),` + `"Not included"` + `)` + `)` + `)`; formulas.push([ `=${sourceRow}`, procedureFormula, durationFormula, assessmentFormula ]); } /******************************************************************** * WRITE PATIENT FORMULAS ********************************************************************/ output .getRange( 6, 1, formulas.length, 4 ) .setFormulas( formulas ); const auditLastRow = 5 + formulas.length; output .getRange( 5, 1, formulas.length + 1, 4 ) .setBorder( true, true, true, true, true, true ); /******************************************************************** * * FOUR-SECTION SUMMARY * ********************************************************************/ output .getRange("F5:H10") .setValues([ [ "Category", "Patient Count (n)", "%" ], [ "Total Hip – Appropriate Duration", "", "" ], [ "Total Hip – Inappropriate Duration", "", "" ], [ "Total Knee – Appropriate Duration", "", "" ], [ "Total Knee – Inappropriate Duration", "", "" ], [ "TOTAL", "", "" ] ]); output .getRange("F5:H5") .setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#EAF2F8"); /******************************************************************** * HIP APPROPRIATE ********************************************************************/ output .getRange("G6") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Hip",` + `$D$6:$D$${auditLastRow},"Appropriate"` + `)` ); /******************************************************************** * HIP INAPPROPRIATE ********************************************************************/ output .getRange("G7") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Hip",` + `$D$6:$D$${auditLastRow},"Inappropriate"` + `)` ); /******************************************************************** * KNEE APPROPRIATE ********************************************************************/ output .getRange("G8") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Knee",` + `$D$6:$D$${auditLastRow},"Appropriate"` + `)` ); /******************************************************************** * KNEE INAPPROPRIATE ********************************************************************/ output .getRange("G9") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Knee",` + `$D$6:$D$${auditLastRow},"Inappropriate"` + `)` ); /******************************************************************** * TOTAL ********************************************************************/ output .getRange("G10") .setFormula( "=SUM(G6:G9)" ); /******************************************************************** * PERCENTAGES * * SAME DENOMINATOR FOR ALL FOUR CATEGORIES ********************************************************************/ output .getRange("H6") .setFormula( "=IFERROR(G6/$G$10,0)" ); output .getRange("H7") .setFormula( "=IFERROR(G7/$G$10,0)" ); output .getRange("H8") .setFormula( "=IFERROR(G8/$G$10,0)" ); output .getRange("H9") .setFormula( "=IFERROR(G9/$G$10,0)" ); output .getRange("H10") .setFormula( "=SUM(H6:H9)" ); output .getRange("H6:H10") .setNumberFormat( "0.0%" ); output .getRange("F5:H10") .setBorder( true, true, true, true, true, true ); output .getRange("F10:H10") .setFontWeight( "bold" ); /******************************************************************** * * PIE SOURCE * * Legend contains n. * * Example: * * Hip – Appropriate (n=9) * ********************************************************************/ output .getRange("J5:K9") .setValues([ [ "Category", "Patients" ], [ "", "" ], [ "", "" ], [ "", "" ], [ "", "" ] ]); output .getRange("J5:K5") .setFontWeight("bold") .setBackground("#EAF2F8"); output .getRange("J6") .setFormula( `="Hip – Appropriate (n="&G6&")"` ); output .getRange("K6") .setFormula( "=G6" ); output .getRange("J7") .setFormula( `="Hip – Inappropriate (n="&G7&")"` ); output .getRange("K7") .setFormula( "=G7" ); output .getRange("J8") .setFormula( `="Knee – Appropriate (n="&G8&")"` ); output .getRange("K8") .setFormula( "=G8" ); output .getRange("J9") .setFormula( `="Knee – Inappropriate (n="&G9&")"` ); output .getRange("K9") .setFormula( "=G9" ); output .getRange("J5:K9") .setBorder( true, true, true, true, true, true ); /******************************************************************** * VALIDATION ********************************************************************/ output .getRange("F12:G15") .setValues([ [ "Validation", "Result" ], [ "Hip Patients", "" ], [ "Knee Patients", "" ], [ "Total Percentage", "" ] ]); output .getRange("F12:G12") .setFontWeight("bold") .setBackground("#EAF2F8"); output .getRange("G13") .setFormula( `=COUNTIF($B$6:$B$${auditLastRow},"Hip")` ); output .getRange("G14") .setFormula( `=COUNTIF($B$6:$B$${auditLastRow},"Knee")` ); output .getRange("G15") .setFormula( "=SUM(H6:H9)" ); output .getRange("G15") .setNumberFormat( "0.0%" ); output .getRange("F12:G15") .setBorder( true, true, true, true, true, true ); /******************************************************************** * COLUMN WIDTHS ********************************************************************/ output.setColumnWidth( 1, 90 ); output.setColumnWidth( 2, 120 ); output.setColumnWidth( 3, 120 ); output.setColumnWidth( 4, 150 ); output.setColumnWidth( 6, 300 ); output.setColumnWidth( 7, 125 ); output.setColumnWidth( 8, 100 ); output.setColumnWidth( 10, 250 ); output.setColumnWidth( 11, 100 ); /******************************************************************** * FORCE FORMULAS TO CALCULATE BEFORE CHART ********************************************************************/ SpreadsheetApp.flush(); Utilities.sleep( 1500 ); SpreadsheetApp.flush(); /******************************************************************** * CREATE PIE CHART ********************************************************************/ const chart = output .newChart() .asPieChart() .addRange( output.getRange( "J5:K9" ) ) .setNumHeaders( 1 ) .setPosition( 18, 6, 0, 0 ) .setOption( "title", "Apixaban Duration Appropriateness by Surgery Type" ) /**************************************************************** * PERCENTAGE DIRECTLY ON PIE ****************************************************************/ .setOption( "pieSliceText", "percentage" ) /**************************************************************** * SHOW VERY SMALL SLICE ****************************************************************/ .setOption( "sliceVisibilityThreshold", 0 ) .setOption( "legend", { position: "right" } ) .setOption( "width", 950 ) .setOption( "height", 600 ) .build(); output.insertChart( chart ); SpreadsheetApp.flush(); /******************************************************************** * GO TO OUTPUT SHEET ********************************************************************/ ss.setActiveSheet( output ); SpreadsheetApp .getUi() .alert( "4-section pie chart created successfully.\n\n" + "New sheet:\n" + OUTPUT_SHEET + "\n\n" + "The legend displays patient count (n), and the pie displays percentage." ); } /********************************************************************** * * FIND HEADER * **********************************************************************/ function findFourPieHeader_( headers, options ) { const cleanHeaders = headers.map( function(header) { return String( header || "" ) .trim() .toLowerCase() .replace( /\s+/g, " " ); } ); for ( let i = 0; i < options.length; i++ ) { const target = String( options[i] ) .trim() .toLowerCase() .replace( /\s+/g, " " ); const index = cleanHeaders.indexOf( target ); if ( index !== -1 ) { return index + 1; } } return null; } /********************************************************************** * * COLUMN NUMBER → LETTER * **********************************************************************/ function fourPieColumnLetter_( number ) { let result = ""; while ( number > 0 ) { const remainder = ( number - 1 ) % 26; result = String.fromCharCode( 65 + remainder ) + result; number = Math.floor( ( number - 1 ) / 26 ); } return result; }
