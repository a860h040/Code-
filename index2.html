range.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER); return shape; } function addRectangle(slide, x, y, w, h, fillColor, borderColor) { const shape = slide.insertShape(SlidesApp.ShapeType.RECTANGLE, x, y, w, h); } /*************************** TABLES ****************************/ function addMethodsTable(slide, data, x, y, w, h, fontSize) { const table = slide.insertTable(data.length, 2, x, y, w, h); for (let r = 0; SlidesApp.ParagraphAlignment.START : SlidesApp.ParagraphAlignment.CENTER ); } } return table; } /*************************** CHARTS ****************************/ function drawAdherenceChart(slide, x, y, w, h, S) { addText(slide, 'Figure 1. range.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER); } currentX += segW; }); } function addLegend(slide, x, y, color, label, S) { const sq = 7 * S; addRectangle(slide, x, y, sq, sq, color, null);
