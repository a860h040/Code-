const SETTINGS = { SHEET_NAME: "Calendar Feed", DAYS_BACK: 30, DAYS_FORWARD: 365, // true = import calendars checked/visible in Google Calendar // false = import every calendar you own or subscribe to ONLY_SELECTED_CALENDARS: true }; /** * Run this function FIRST. */ function setupCalendarFeed() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error( "Open this script from Google Sheets using Extensions > Apps Script." ); } PropertiesService.getScriptProperties().setProperty( "SPREADSHEET_ID", spreadsheet.getId() ); removeCalendarTriggers(); ScriptApp.newTrigger("syncCalendarToSheet") .timeBased() .everyMinutes(15) .create(); syncCalendarToSheet(); SpreadsheetApp.getUi().alert( "Calendar connection completed. Open the Calendar Feed tab." ); } /** * Pulls events from Google Calendar into the sheet. */ function syncCalendarToSheet() { const spreadsheet = getSpreadsheet(); let sheet = spreadsheet.getSheetByName(SETTINGS.SHEET_NAME); if (!sheet) { sheet = spreadsheet.insertSheet(SETTINGS.SHEET_NAME); } sheet.clear(); sheet.getRange("A1").setValue("Loading calendar events..."); SpreadsheetApp.flush(); try { const startDate = new Date(); startDate.setDate(startDate.getDate() - SETTINGS.DAYS_BACK); startDate.setHours(0, 0, 0, 0); const endDate = new Date(); endDate.setDate(endDate.getDate() + SETTINGS.DAYS_FORWARD); endDate.setHours(23, 59, 59, 999); let calendars = CalendarApp.getAllCalendars(); if (SETTINGS.ONLY_SELECTED_CALENDARS) { calendars = calendars.filter(calendar => calendar.isSelected()); } if (calendars.length === 0) { throw new Error( "No selected calendars were found. Open Google Calendar and check the calendar boxes on the left." ); } const rows = []; calendars.forEach(calendar => { const events = calendar.getEvents(startDate, endDate); events.forEach(event => { let guests = ""; try { guests = event .getGuestList() .map(guest => guest.getEmail()) .filter(Boolean) .join(", "); } catch (guestError) { guests = ""; } rows.push([ calendar.getName(), event.getTitle() || "(No title)", event.getStartTime(), event.getEndTime(), event.isAllDayEvent() ? "Yes" : "No", event.getLocation() || "", event.getDescription() || "", guests, event.getId() ]); }); }); rows.sort((a, b) => a[2] - b[2]); const headers = [ "Calendar", "Event", "Start", "End", "All Day", "Location", "Description", "Guests", "Event ID" ]; sheet.clear(); sheet .getRange(1, 1, 1, headers.length) .setValues([headers]) .setFontWeight("bold") .setBackground("#1a73e8") .setFontColor("#ffffff"); if (rows.length > 0) { sheet .getRange(2, 1, rows.length, headers.length) .setValues(rows); sheet .getRange(2, 3, rows.length, 2) .setNumberFormat("MM/dd/yyyy hh:mm AM/PM"); sheet .getRange(1, 1, rows.length + 1, headers.length) .createFilter(); } else { sheet.getRange("A2").setValue( "The connection worked, but no events were found within the selected date range." ); } sheet.setFrozenRows(1); sheet.autoResizeColumns(1, headers.length); sheet.setColumnWidth(2, 250); sheet.setColumnWidth(6, 220); sheet.setColumnWidth(7, 350); sheet.setColumnWidth(8, 280); if (rows.length > 0) { sheet .getRange(2, 6, rows.length, 3) .setWrap(true); } sheet.getRange("K1").setValue("Last synchronized"); sheet.getRange("K2").setValue(new Date()); sheet.getRange("K2").setNumberFormat( "MM/dd/yyyy hh:mm:ss AM/PM" ); sheet.getRange("K4").setValue("Calendars checked"); sheet.getRange("K5").setValue(calendars.length); sheet.getRange("K7").setValue("Events found"); sheet.getRange("K8").setValue(rows.length); } catch (error) { sheet.clear(); sheet.getRange("A1").setValue("CALENDAR ERROR"); sheet.getRange("A1").setFontWeight("bold"); sheet.getRange("A2").setValue(error.message); sheet.getRange("A4").setValue( "Open Apps Script > Executions to view the complete error." ); throw error; } } /** * Shows every calendar the script can access. */ function testCalendarAccess() { const spreadsheet = getSpreadsheet(); let sheet = spreadsheet.getSheetByName("Calendar Test"); if (!sheet) { sheet = spreadsheet.insertSheet("Calendar Test"); } const calendars = CalendarApp.getAllCalendars(); const rows = calendars.map(calendar => [ calendar.getName(), calendar.getId(), calendar.isSelected() ? "Yes" : "No", calendar.isHidden() ? "Yes" : "No", calendar.isOwnedByMe() ? "Yes" : "No" ]); sheet.clear(); sheet.getRange(1, 1, 1, 5).setValues([[ "Calendar Name", "Calendar ID", "Selected", "Hidden", "Owned By Me" ]]); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, 5).setValues(rows); } sheet.setFrozenRows(1); sheet.autoResizeColumns(1, 5); } /** * Opens the correct spreadsheet during automatic trigger runs. */ function getSpreadsheet() { const spreadsheetId = PropertiesService.getScriptProperties() .getProperty("SPREADSHEET_ID"); if (spreadsheetId) { return SpreadsheetApp.openById(spreadsheetId); } const activeSpreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!activeSpreadsheet) { throw new Error( "Spreadsheet connection is missing. Run setupCalendarFeed first." ); } return activeSpreadsheet; } /** * Removes existing synchronization triggers. */ function removeCalendarTriggers() { ScriptApp.getProjectTriggers().forEach(trigger => { if (trigger.getHandlerFunction() === "syncCalendarToSheet") { ScriptApp.deleteTrigger(trigger); } }); }
