/********************************************************************** * 12 APIXABAN MUE * TWO SIDE-BY-SIDE PIE CHARTS * * SOURCE: * Sheet1 * * OUTPUT: * MUE 2-Pie Duration * * LEFT PIE: * Total Hip Arthroplasty * - Appropriate Duration * - Inappropriate Duration * * RIGHT PIE: * Total Knee Arthroplasty * - Appropriate Duration * - Inappropriate Duration * * STANDARD: * Hip = 35 days * Knee = 12 days * * IMPORTANT: * ------------------------------------------------------------ * - Sheet1 is NOT modified. * - All calculations are formula-driven. * - Each pie equals 100% of that procedure population. * - Legend contains patient count (n). * - Pie slices display percentage. * * RUN: * buildTwoDurationPieCharts **********************************************************************/ function buildTwoDurationPieCharts() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const SOURCE_SHEET = "Sheet1"; const OUTPUT_SHEET = "MUE 2-Pie Duration"; const HIP_DURATION = 35; const KNEE_DURATION = 12; /******************************************************************** * GET SOURCE ********************************************************************/ const source = ss.getSheetByName( SOURCE_SHEET ); if (!source) { SpreadsheetApp .getUi() .alert( 'Could not find sheet "' + SOURCE_SHEET + '".' ); return; } const lastRow = source.getLastRow(); const lastColumn = source.getLastColumn(); if (lastRow < 2) { SpreadsheetApp .getUi() .alert( "Sheet1 does not contain patient data." ); return; } /******************************************************************** * READ HEADERS ********************************************************************/ const headers = source .getRange( 1, 1, 1, lastColumn ) .getDisplayValues()[0]; /******************************************************************** * FIND SURGERY COLUMN ********************************************************************/ const surgeryColumn = findTwoPieHeader_( headers, [ "Type of surgery", "Surgery Type", "Procedure Type", "Procedure" ] ); /******************************************************************** * FIND DURATION COLUMN ********************************************************************/ const durationColumn = findTwoPieHeader_( headers, [ "Number of outpatient days supplied (total treatment duration)", "Total treatment duration", "Total Treatment Duration", "Total treatment duration (days)", "Treatment duration", "Total Duration", "Duration" ] ); if (!surgeryColumn) { SpreadsheetApp .getUi() .alert( 'Could not find the "Type of surgery" column.' ); return; } if (!durationColumn) { SpreadsheetApp .getUi() .alert( "Could not find the treatment duration column." ); return; } /******************************************************************** * COLUMN LETTERS ********************************************************************/ const surgeryLetter = twoPieColumnLetter_( surgeryColumn ); const durationLetter = twoPieColumnLetter_( durationColumn ); const safeSource = "'" + source .getName() .replace( /'/g, "''" ) + "'"; /******************************************************************** * CREATE / RESET OUTPUT SHEET ********************************************************************/ let output = ss.getSheetByName( OUTPUT_SHEET ); if (!output) { output = ss.insertSheet( OUTPUT_SHEET ); } else { /****************************************************************** * DELETE OLD CHARTS ******************************************************************/ output .getCharts() .forEach( function(chart) { output.removeChart( chart ); } ); /****************************************************************** * BREAK OLD MERGED CELLS ******************************************************************/ output .getDataRange() .getMergedRanges() .forEach( function(range) { range.breakApart(); } ); output.clear(); } /******************************************************************** * MAIN TITLE ********************************************************************/ output .getRange("A1:L1") .merge() .setValue( "Apixaban Duration Appropriateness by Surgery Type" ) .setFontSize(16) .setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#D9EAF7"); output .getRange("A2:L2") .merge() .setValue( "Total hip and total knee arthroplasty are displayed separately. " + "Each pie represents 100% of patients within that procedure type." ) .setHorizontalAlignment("center") .setWrap(true); output .getRange("A3:L3") .merge() .setValue( "Appropriate duration: Total Hip = 35 days; Total Knee = 12 days." ) .setHorizontalAlignment("center") .setFontStyle("italic"); /******************************************************************** * * PATIENT-LEVEL AUDIT TABLE * ********************************************************************/ output .getRange("A5:D5") .setValues([ [ "Source Row", "Procedure", "Duration (days)", "Duration Assessment" ] ]) .setFontWeight("bold") .setBackground("#EAF2F8") .setHorizontalAlignment("center"); const formulas = []; for ( let sourceRow = 2; sourceRow <= lastRow; sourceRow++ ) { const outputRow = sourceRow + 4; const surgeryRef = `${safeSource}!${surgeryLetter}${sourceRow}`; const durationRef = `${safeSource}!${durationLetter}${sourceRow}`; /****************************************************************** * PROCEDURE ******************************************************************/ const procedureFormula = `=IF(` + `OR(` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total joint.*hip"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total hip"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"hip arthroplasty"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"(^|[^a-z])tha([^a-z]|$)")` + `),` + `"Hip",` + `IF(` + `OR(` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total joint.*knee"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"total knee"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"knee arthroplasty"),` + `REGEXMATCH(LOWER(TO_TEXT(${surgeryRef})),"(^|[^a-z])tka([^a-z]|$)")` + `),` + `"Knee",` + `"Not included"` + `)` + `)`; /****************************************************************** * ACTUAL DURATION ******************************************************************/ const durationFormula = `=IFERROR(` + `VALUE(` + `REGEXEXTRACT(` + `TO_TEXT(${durationRef}),` + `"[0-9]+\\.?[0-9]*"` + `)` + `),` + `""` + `)`; /****************************************************************** * APPROPRIATE / INAPPROPRIATE * * HIP: * 35 days = Appropriate * * KNEE: * 12 days = Appropriate * * Any other duration = Inappropriate ******************************************************************/ const assessmentFormula = `=IF(` + `B${outputRow}="Hip",` + `IF(` + `C${outputRow}=${HIP_DURATION},` + `"Appropriate",` + `"Inappropriate"` + `),` + `IF(` + `B${outputRow}="Knee",` + `IF(` + `C${outputRow}=${KNEE_DURATION},` + `"Appropriate",` + `"Inappropriate"` + `),` + `"Not included"` + `)` + `)`; formulas.push([ `=${sourceRow}`, procedureFormula, durationFormula, assessmentFormula ]); } /******************************************************************** * WRITE AUDIT FORMULAS ********************************************************************/ output .getRange( 6, 1, formulas.length, 4 ) .setFormulas( formulas ); const auditLastRow = 5 + formulas.length; output .getRange( 5, 1, formulas.length + 1, 4 ) .setBorder( true, true, true, true, true, true ); /******************************************************************** * * HIP SUMMARY * ********************************************************************/ output .getRange("F5:H8") .setValues([ [ "Total Hip Arthroplasty", "n", "%" ], [ "Appropriate Duration", "", "" ], [ "Inappropriate Duration", "", "" ], [ "Total", "", "" ] ]); output .getRange("F5:H5") .setFontWeight("bold") .setBackground("#EAF2F8") .setHorizontalAlignment("center"); /******************************************************************** * HIP APPROPRIATE ********************************************************************/ output .getRange("G6") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Hip",` + `$D$6:$D$${auditLastRow},"Appropriate"` + `)` ); /******************************************************************** * HIP INAPPROPRIATE ********************************************************************/ output .getRange("G7") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Hip",` + `$D$6:$D$${auditLastRow},"Inappropriate"` + `)` ); /******************************************************************** * HIP TOTAL ********************************************************************/ output .getRange("G8") .setFormula( "=SUM(G6:G7)" ); output .getRange("H6") .setFormula( "=IFERROR(G6/$G$8,0)" ); output .getRange("H7") .setFormula( "=IFERROR(G7/$G$8,0)" ); output .getRange("H8") .setFormula( "=SUM(H6:H7)" ); output .getRange("H6:H8") .setNumberFormat( "0.0%" ); output .getRange("F5:H8") .setBorder( true, true, true, true, true, true ); /******************************************************************** * * KNEE SUMMARY * ********************************************************************/ output .getRange("J5:L8") .setValues([ [ "Total Knee Arthroplasty", "n", "%" ], [ "Appropriate Duration", "", "" ], [ "Inappropriate Duration", "", "" ], [ "Total", "", "" ] ]); output .getRange("J5:L5") .setFontWeight("bold") .setBackground("#EAF2F8") .setHorizontalAlignment("center"); /******************************************************************** * KNEE APPROPRIATE ********************************************************************/ output .getRange("K6") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Knee",` + `$D$6:$D$${auditLastRow},"Appropriate"` + `)` ); /******************************************************************** * KNEE INAPPROPRIATE ********************************************************************/ output .getRange("K7") .setFormula( `=COUNTIFS(` + `$B$6:$B$${auditLastRow},"Knee",` + `$D$6:$D$${auditLastRow},"Inappropriate"` + `)` ); /******************************************************************** * KNEE TOTAL ********************************************************************/ output .getRange("K8") .setFormula( "=SUM(K6:K7)" ); output .getRange("L6") .setFormula( "=IFERROR(K6/$K$8,0)" ); output .getRange("L7") .setFormula( "=IFERROR(K7/$K$8,0)" ); output .getRange("L8") .setFormula( "=SUM(L6:L7)" ); output .getRange("L6:L8") .setNumberFormat( "0.0%" ); output .getRange("J5:L8") .setBorder( true, true, true, true, true, true ); /******************************************************************** * * HIP PIE SOURCE * * Legend includes n. * ********************************************************************/ output .getRange("F10:G12") .setValues([ [ "Hip", "Patients" ], [ "", "" ], [ "", "" ] ]); output .getRange("F10:G10") .setFontWeight("bold"); output .getRange("F11") .setFormula( `="Appropriate Duration (n="&G6&")"` ); output .getRange("G11") .setFormula( "=G6" ); output .getRange("F12") .setFormula( `="Inappropriate Duration (n="&G7&")"` ); output .getRange("G12") .setFormula( "=G7" ); /******************************************************************** * * KNEE PIE SOURCE * ********************************************************************/ output .getRange("J10:K12") .setValues([ [ "Knee", "Patients" ], [ "", "" ], [ "", "" ] ]); output .getRange("J10:K10") .setFontWeight("bold"); output .getRange("J11") .setFormula( `="Appropriate Duration (n="&K6&")"` ); output .getRange("K11") .setFormula( "=K6" ); output .getRange("J12") .setFormula( `="Inappropriate Duration (n="&K7&")"` ); output .getRange("K12") .setFormula( "=K7" ); /******************************************************************** * WIDTHS ********************************************************************/ output.setColumnWidth( 1, 85 ); output.setColumnWidth( 2, 120 ); output.setColumnWidth( 3, 120 ); output.setColumnWidth( 4, 150 ); output.setColumnWidth( 6, 240 ); output.setColumnWidth( 7, 90 ); output.setColumnWidth( 8, 90 ); output.setColumnWidth( 10, 240 ); output.setColumnWidth( 11, 90 ); output.setColumnWidth( 12, 90 ); /******************************************************************** * WAIT FOR FORMULAS ********************************************************************/ SpreadsheetApp.flush(); Utilities.sleep( 1200 ); SpreadsheetApp.flush(); /******************************************************************** * * HIP PIE CHART * ********************************************************************/ const hipChart = output .newChart() .asPieChart() .addRange( output.getRange( "F10:G12" ) ) .setNumHeaders( 1 ) .setPosition( 15, 5, 0, 0 ) .setOption( "title", "Total Hip Arthroplasty – Duration Appropriateness" ) .setOption( "pieSliceText", "percentage" ) .setOption( "sliceVisibilityThreshold", 0 ) .setOption( "legend", { position: "bottom" } ) .setOption( "width", 650 ) .setOption( "height", 450 ) .build(); output.insertChart( hipChart ); /******************************************************************** * * KNEE PIE CHART * ********************************************************************/ const kneeChart = output .newChart() .asPieChart() .addRange( output.getRange( "J10:K12" ) ) .setNumHeaders( 1 ) .setPosition( 15, 11, 0, 0 ) .setOption( "title", "Total Knee Arthroplasty – Duration Appropriateness" ) .setOption( "pieSliceText", "percentage" ) .setOption( "sliceVisibilityThreshold", 0 ) .setOption( "legend", { position: "bottom" } ) .setOption( "width", 650 ) .setOption( "height", 450 ) .build(); output.insertChart( kneeChart ); SpreadsheetApp.flush(); /******************************************************************** * OPEN OUTPUT ********************************************************************/ ss.setActiveSheet( output ); SpreadsheetApp .getUi() .alert( "Two pie charts created successfully.\n\n" + "Left: Total Hip Arthroplasty\n" + "Right: Total Knee Arthroplasty\n\n" + "Each pie equals 100% of its procedure population." ); } /********************************************************************** * * FIND HEADER * **********************************************************************/ function findTwoPieHeader_( headers, possibilities ) { const normalized = headers.map( function(header) { return String( header || "" ) .trim() .toLowerCase() .replace( /\s+/g, " " ); } ); for ( let i = 0; i < possibilities.length; i++ ) { const target = String( possibilities[i] ) .trim() .toLowerCase() .replace( /\s+/g, " " ); const index = normalized.indexOf( target ); if ( index !== -1 ) { return index + 1; } } return null; } /********************************************************************** * * COLUMN NUMBER → LETTER * **********************************************************************/ function twoPieColumnLetter_( number ) { let result = ""; while ( number > 0 ) { const remainder = ( number - 1 ) % 26; result = String.fromCharCode( 65 + remainder ) + result; number = Math.floor( ( number - 1 ) / 26 ); } return result; }
