const SETTINGS = { SHEET_NAME: "Calendar Feed", // How many days to import DAYS_BACK: 30, DAYS_FORWARD: 365, /* * false = import all calendars available to your Google account * true = import only calendars checked/visible in Google Calendar */ ONLY_SELECTED_CALENDARS: false, /* * Optional: enter specific Calendar IDs below. * Leave empty [] to import all accessible calendars. * * Example: * CALENDAR_IDS: [ * "your-email@gmail.com", * "calendar-id@group.calendar.google.com" * ] */ CALENDAR_IDS: [], // Automatic refresh interval REFRESH_EVERY_MINUTES: 15 }; /** * Creates the custom menu when the Google Sheet opens. */ function onOpen() { SpreadsheetApp.getUi() .createMenu("Calendar Feed") .addItem("Set up calendar feed", "setupCalendarFeed") .addItem("Sync calendar now", "syncCalendarToSheet") .addSeparator() .addItem("Test calendar access", "testCalendarAccess") .addItem("Remove automatic sync", "removeAutomaticSync") .addToUi(); } /** * Run this function first. * * It: * 1. Connects the script to the spreadsheet * 2. Imports calendar events * 3. Creates an automatic 15-minute trigger */ function setupCalendarFeed() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error( "Open the Google Sheet first, then go to Extensions > Apps Script." ); } PropertiesService.getScriptProperties().setProperty( "SPREADSHEET_ID", spreadsheet.getId() ); // Test the connection and import events first. syncCalendarToSheet(); // Remove old duplicate triggers. removeCalendarTriggers(); // Create a new automatic trigger. ScriptApp.newTrigger("syncCalendarToSheet") .timeBased() .everyMinutes(SETTINGS.REFRESH_EVERY_MINUTES) .create(); SpreadsheetApp.getUi().alert( "Calendar Feed setup is complete.\n\n" + "Your calendar will refresh approximately every " + SETTINGS.REFRESH_EVERY_MINUTES + " minutes." ); } /** * Imports Google Calendar events into the Google Sheet. */ function syncCalendarToSheet() { const spreadsheet = getConnectedSpreadsheet(); const sheet = getOrCreateSheet( spreadsheet, SETTINGS.SHEET_NAME ); // Prevent two triggers from updating the sheet simultaneously. const lock = LockService.getScriptLock(); try { lock.waitLock(30000); } catch (error) { console.log( "Another calendar synchronization is already running." ); return; } try { resetSheet(sheet); sheet.getRange("A1").setValue( "Loading calendar events..." ); SpreadsheetApp.flush(); const dateRange = getDateRange(); const calendars = getCalendarsToImport(); if (calendars.length === 0) { throw new Error( "No calendars were found. Run testCalendarAccess to check which calendars this script can access." ); } const rows = []; calendars.forEach(calendar => { try { const events = calendar.getEvents( dateRange.startDate, dateRange.endDate ); events.forEach(event => { rows.push( createEventRow(calendar, event) ); }); } catch (calendarError) { console.log( "Could not read calendar: " + calendar.getName() + ". Error: " + calendarError.message ); } }); // Sort events by start date and time. rows.sort((firstRow, secondRow) => { return firstRow[4] - secondRow[4]; }); writeCalendarFeed( sheet, rows, calendars.length ); spreadsheet.toast( rows.length + " calendar events imported from " + calendars.length + " calendar(s).", "Calendar Feed", 6 ); } catch (error) { showErrorInSheet(sheet, error); throw error; } finally { lock.releaseLock(); } } /** * Returns the dates that should be searched. */ function getDateRange() { const startDate = new Date(); startDate.setDate( startDate.getDate() - SETTINGS.DAYS_BACK ); startDate.setHours(0, 0, 0, 0); const endDate = new Date(); endDate.setDate( endDate.getDate() + SETTINGS.DAYS_FORWARD ); endDate.setHours(23, 59, 59, 999); return { startDate: startDate, endDate: endDate }; } /** * Determines which calendars should be imported. */ function getCalendarsToImport() { // Use specific Calendar IDs when provided. if ( Array.isArray(SETTINGS.CALENDAR_IDS) && SETTINGS.CALENDAR_IDS.length > 0 ) { return SETTINGS.CALENDAR_IDS .map(calendarId => { return CalendarApp.getCalendarById( String(calendarId).trim() ); }) .filter(calendar => calendar !== null); } let calendars = CalendarApp.getAllCalendars(); if (SETTINGS.ONLY_SELECTED_CALENDARS) { calendars = calendars.filter(calendar => { return calendar.isSelected(); }); } return calendars; } /** * Converts one calendar event into a spreadsheet row. */ function createEventRow(calendar, event) { const startTime = event.getStartTime(); const endTime = event.getEndTime(); let guests = ""; let creators = ""; let myStatus = ""; let recurring = "No"; let lastUpdated = ""; try { guests = event .getGuestList() .map(guest => guest.getEmail()) .filter(email => email) .join(", "); } catch (error) { guests = ""; } try { creators = event .getCreators() .filter(creator => creator) .join(", "); } catch (error) { creators = ""; } try { myStatus = String(event.getMyStatus()); } catch (error) { myStatus = ""; } try { recurring = event.isRecurringEvent() ? "Yes" : "No"; } catch (error) { recurring = "No"; } try { lastUpdated = event.getLastUpdated(); } catch (error) { lastUpdated = ""; } const eventId = event.getId(); // A recurring event may use the same Event ID for multiple dates. const eventKey = eventId + "|" + startTime.toISOString(); return [ eventKey, eventId, calendar.getName(), event.getTitle() || "(No title)", startTime, endTime, event.isAllDayEvent() ? "Yes" : "No", event.getLocation() || "", event.getDescription() || "", creators, guests, myStatus, recurring, lastUpdated ]; } /** * Writes all calendar events into the sheet. */ function writeCalendarFeed( sheet, rows, calendarCount ) { resetSheet(sheet); const headers = [ "Event Key", "Event ID", "Calendar", "Event Title", "Start Date and Time", "End Date and Time", "All-Day Event", "Location", "Description", "Created By", "Guests", "My Status", "Recurring Event", "Last Updated" ]; sheet .getRange(1, 1, 1, headers.length) .setValues([headers]) .setFontWeight("bold") .setBackground("#1a73e8") .setFontColor("#ffffff") .setHorizontalAlignment("center"); if (rows.length > 0) { sheet .getRange( 2, 1, rows.length, headers.length ) .setValues(rows); // Start and end date columns. sheet .getRange(2, 5, rows.length, 2) .setNumberFormat( "MM/dd/yyyy hh:mm AM/PM" ); // Last updated column. sheet .getRange(2, 14, rows.length, 1) .setNumberFormat( "MM/dd/yyyy hh:mm AM/PM" ); sheet .getRange( 1, 1, rows.length + 1, headers.length ) .createFilter(); sheet .getRange( 2, 1, rows.length, headers.length ) .setVerticalAlignment("middle"); sheet .getRange(2, 8, rows.length, 4) .setWrap(true); } else { sheet .getRange("A2") .setValue( "The calendar connection worked, but no events were found within the selected date range." ); sheet .getRange("A2:N2") .merge() .setWrap(true); } sheet.setFrozenRows(1); sheet.setFrozenColumns(1); setColumnWidths(sheet); // Synchronization information. sheet.getRange("P1").setValue( "Last synchronized" ); sheet.getRange("P1").setFontWeight("bold"); sheet.getRange("P2").setValue(new Date()); sheet .getRange("P2") .setNumberFormat( "MM/dd/yyyy hh:mm:ss AM/PM" ); sheet.getRange("P4").setValue( "Calendars checked" ); sheet.getRange("P4").setFontWeight("bold"); sheet.getRange("P5").setValue(calendarCount); sheet.getRange("P7").setValue( "Events found" ); sheet.getRange("P7").setFontWeight("bold"); sheet.getRange("P8").setValue(rows.length); sheet.getRange("P10").setValue( "Date range" ); sheet.getRange("P10").setFontWeight("bold"); sheet.getRange("P11").setValue( SETTINGS.DAYS_BACK + " days back through " + SETTINGS.DAYS_FORWARD + " days forward" ); sheet.setColumnWidth(16, 210); } /** * Sets the width of each calendar column. */ function setColumnWidths(sheet) { sheet.setColumnWidth(1, 290); sheet.setColumnWidth(2, 250); sheet.setColumnWidth(3, 180); sheet.setColumnWidth(4, 260); sheet.setColumnWidth(5, 180); sheet.setColumnWidth(6, 180); sheet.setColumnWidth(7, 110); sheet.setColumnWidth(8, 220); sheet.setColumnWidth(9, 350); sheet.setColumnWidth(10, 220); sheet.setColumnWidth(11, 300); sheet.setColumnWidth(12, 130); sheet.setColumnWidth(13, 130); sheet.setColumnWidth(14, 180); } /** * Shows every calendar that the script can access. */ function testCalendarAccess() { const spreadsheet = getConnectedSpreadsheet(); const sheet = getOrCreateSheet( spreadsheet, "Calendar Test" ); resetSheet(sheet); const calendars = CalendarApp.getAllCalendars(); const headers = [ "Calendar Name", "Calendar ID", "Selected", "Hidden", "Owned By Me", "Description" ]; const rows = calendars.map(calendar => { let description = ""; try { description = calendar.getDescription() || ""; } catch (error) { description = ""; } return [ calendar.getName(), calendar.getId(), calendar.isSelected() ? "Yes" : "No", calendar.isHidden() ? "Yes" : "No", calendar.isOwnedByMe() ? "Yes" : "No", description ]; }); sheet .getRange(1, 1, 1, headers.length) .setValues([headers]) .setFontWeight("bold") .setBackground("#1a73e8") .setFontColor("#ffffff"); if (rows.length > 0) { sheet .getRange(2, 1, rows.length, headers.length) .setValues(rows); sheet .getRange(1, 1, rows.length + 1, headers.length) .createFilter(); } else { sheet.getRange("A2").setValue( "No calendars are accessible with this Google account." ); } sheet.setFrozenRows(1); sheet.setColumnWidth(1, 230); sheet.setColumnWidth(2, 420); sheet.setColumnWidth(3, 100); sheet.setColumnWidth(4, 100); sheet.setColumnWidth(5, 120); sheet.setColumnWidth(6, 350); spreadsheet.toast( calendars.length + " accessible calendar(s) found.", "Calendar Test", 5 ); } /** * Opens the correct spreadsheet during automatic trigger runs. */ function getConnectedSpreadsheet() { const spreadsheetId = PropertiesService .getScriptProperties() .getProperty("SPREADSHEET_ID"); if (spreadsheetId) { return SpreadsheetApp.openById( spreadsheetId ); } const activeSpreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!activeSpreadsheet) { throw new Error( "The spreadsheet is not connected. Run setupCalendarFeed first." ); } PropertiesService.getScriptProperties() .setProperty( "SPREADSHEET_ID", activeSpreadsheet.getId() ); return activeSpreadsheet; } /** * Gets an existing sheet or creates a new one. */ function getOrCreateSheet( spreadsheet, sheetName ) { let sheet = spreadsheet.getSheetByName(sheetName); if (!sheet) { sheet = spreadsheet.insertSheet( sheetName ); } return sheet; } /** * Removes the filter and clears the sheet. * * This fixes: * "You can't create a filter in a sheet that already has a filter." */ function resetSheet(sheet) { const filter = sheet.getFilter(); if (filter) { filter.remove(); } const fullRange = sheet.getDataRange(); try { fullRange.breakApart(); } catch (error) { console.log( "No merged cells needed to be separated." ); } sheet.clear(); } /** * Displays errors directly inside the Calendar Feed sheet. */ function showErrorInSheet(sheet, error) { try { resetSheet(sheet); sheet.getRange("A1").setValue( "CALENDAR ERROR" ); sheet .getRange("A1") .setFontWeight("bold") .setBackground("#d93025") .setFontColor("#ffffff"); sheet.getRange("A2").setValue( error.message || String(error) ); sheet.getRange("A4").setValue( "Open Apps Script > Executions to see the complete error." ); sheet.setColumnWidth(1, 650); sheet.getRange("A1:A4").setWrap(true); } catch (displayError) { console.log(displayError); } } /** * Removes existing calendar synchronization triggers. */ function removeCalendarTriggers() { const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(trigger => { if ( trigger.getHandlerFunction() === "syncCalendarToSheet" ) { ScriptApp.deleteTrigger(trigger); } }); } /** * Removes automatic synchronization. */ function removeAutomaticSync() { removeCalendarTriggers(); SpreadsheetApp.getUi().alert( "Automatic Calendar Feed synchronization has been removed." ); }
