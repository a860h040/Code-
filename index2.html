/******************************************************************** * APIXABAN MUE POSTER BUILDER * --------------------------------------------------------------- * Builds a one-slide research poster similar to the example: *2 * LEFT: * Background * Objective * Methods * * CENTER: * Results * Baseline Characteristics table * Apixaban Use table * Figure 1 - Regimen Adherence * Figure 2 - Duration Appropriateness by Procedure * * RIGHT: * Discussion * Conclusion * References * * IMPORTANT: * This script CLEARS slide 1 before rebuilding it. ********************************************************************/ function buildApixabanPoster() { const presentation = SlidesApp.getActivePresentation(); const slides = presentation.getSlides(); if (slides.length === 0) { presentation.appendSlide(SlidesApp.PredefinedLayout.BLANK); } const slide = presentation.getSlides()[0]; // ------------------------------------------------------------ // CLEAR THE SLIDE // ------------------------------------------------------------ const oldElements = slide.getPageElements(); for (let i = oldElements.length - 1; i >= 0; i--) { oldElements[i].remove(); } // ------------------------------------------------------------ // PAGE SIZE // ------------------------------------------------------------ const W = presentation.getPageWidth(); const H = presentation.getPageHeight(); // ------------------------------------------------------------ // COLORS // ------------------------------------------------------------ const BLUE = "#0C4C8A"; const LIGHT_BLUE = "#EAF2F8"; const MED_BLUE = "#D7E6F3"; const DARK = "#1F1F1F"; const GRAY = "#666666"; const WHITE = "#FFFFFF"; // Chart colors const APPROPRIATE = "#4285F4"; const PROLONGED = "#EA4335"; const ABBREVIATED = "#F4B400"; // ------------------------------------------------------------ // USER INFORMATION // Change these as needed // ------------------------------------------------------------ const AUTHOR = "Alhamza Hamza, PharmD"; const COAUTHORS = "[Preceptor Name, PharmD, Credentials]"; const AFFILIATION = "Ascension St. Vincent’s, Jacksonville, Florida"; const CONTACT = "Alhamza Hamza, PharmD\n" + "Ascension St. Vincent’s Riverside\n" + "Department of Pharmacy\n" + "1 Shircliff Way, Jacksonville, FL 32204\n" + "[Email Address]"; const TITLE = "Evaluation of Apixaban Dosing and Duration for Venous " + "Thromboembolism Prophylaxis Following Total Hip or Knee Arthroplasty"; // ------------------------------------------------------------ // LAYOUT // ------------------------------------------------------------ const margin = W * 0.025; const topArea = H * 0.16; const bottomMargin = H * 0.025; const usableW = W - (2 * margin); const leftW = usableW * 0.245; const centerW = usableW * 0.49; const rightW = usableW * 0.245; const gap = usableW * 0.01; const leftX = margin; const centerX = leftX + leftW + gap; const rightX = centerX + centerW + gap; const contentTop = topArea; const contentH = H - contentTop - bottomMargin; // ============================================================ // TOP HEADER // ============================================================ // ------------------------------------------------------------ // LEFT: ORGANIZATION / CONTACT // ------------------------------------------------------------ addText( slide, "Ascension St. Vincent’s\nDepartment of Pharmacy", leftX, H * 0.025, leftW, H * 0.055, 14, BLUE, true, SlidesApp.ParagraphAlignment.LEFT ); addText( slide, CONTACT, leftX, H * 0.082, leftW, H * 0.065, 8, GRAY, false, SlidesApp.ParagraphAlignment.LEFT ); // ------------------------------------------------------------ // CENTER TITLE // ------------------------------------------------------------ addText( slide, TITLE, centerX, H * 0.018, centerW, H * 0.075, 24, BLUE, true, SlidesApp.ParagraphAlignment.CENTER ); addText( slide, AUTHOR + " • " + COAUTHORS, centerX, H * 0.095, centerW, H * 0.025, 10, DARK, false, SlidesApp.ParagraphAlignment.CENTER ); addText( slide, AFFILIATION, centerX, H * 0.121, centerW, H * 0.020, 9, DARK, false, SlidesApp.ParagraphAlignment.CENTER ); addText( slide, "QR CODE", rightX + rightW * 0.72, H * 0.068, rightW * 0.25, H * 0.02, 8, BLUE, true, SlidesApp.ParagraphAlignment.CENTER ); // ============================================================ // LEFT COLUMN // ============================================================ let y = contentTop; // ------------------------------------------------------------ // BACKGROUND // ------------------------------------------------------------ addSectionHeader( slide, "Background", leftX, y, leftW, H * 0.032, BLUE ); y += H * 0.037; const backgroundText = "• Patients undergoing total hip arthroplasty (THA) or total knee " + "arthroplasty (TKA) are at increased risk for postoperative venous " + "thromboembolism (VTE), including deep vein thrombosis and pulmonary embolism.\n\n" + "• Apixaban is used for VTE prophylaxis following THA and TKA.\n\n" + "• The FDA-approved apixaban regimen is 2.5 mg orally twice daily for " + "35 days following THA and 12 days following TKA.\n\n" + "• Differences in recommended prophylaxis duration between THA and TKA " + "may contribute to variation in discharge prescribing practices.\n\n" + "• Standardized treatment durations may improve consistency and reduce " + "inappropriate prophylaxis."; addBodyText( slide, backgroundText, leftX, y, leftW, H * 0.28 ); y += H * 0.285; // ------------------------------------------------------------ // OBJECTIVE // ------------------------------------------------------------ addSectionHeader( slide, "Objective", leftX, y, leftW, H * 0.032, BLUE ); y += H * 0.037; const objectiveText = "To evaluate the appropriateness of apixaban dose and total treatment " + "duration for VTE prophylaxis following THA or TKA and identify " + "opportunities to improve discharge prescribing."; addBodyText( slide, objectiveText, leftX, y, leftW, H * 0.10 ); y += H * 0.105; // ------------------------------------------------------------ // METHODS // ------------------------------------------------------------ addSectionHeader( slide, "Methods", leftX, y, leftW, H * 0.032, BLUE ); y += H * 0.038; const methodsData = [ ["Study Design", "Retrospective medication use evaluation"], ["Study Period", "November 1, 2025 – August 1, 2026"], ["Population", "Adults undergoing THA or TKA receiving apixaban for postoperative VTE prophylaxis"], ["Appropriate Dose", "Apixaban 2.5 mg orally twice daily"], ["Appropriate Duration", "THA: 35 days\nTKA: 12 days"], ["Primary Outcome", "Percentage receiving both the recommended dose and duration"] ]; addTwoColumnTable( slide, methodsData, leftX, y, leftW, H * 0.29, BLUE, LIGHT_BLUE ); // ============================================================ // CENTER COLUMN // ============================================================ let cy = contentTop; addSectionHeader( slide, "Results", centerX, cy, centerW, H * 0.032, BLUE ); cy += H * 0.042; // ------------------------------------------------------------ // TABLE 1 // ------------------------------------------------------------ const tableGap = centerW * 0.025; const tableWidth = (centerW - tableGap) / 2; const baselineData = [ ["Characteristic", "Result"], ["Patients included", "81"], ["Median age, years (IQR)", "75 (65–80)"], ["Median BMI, kg/m² (IQR)", "29.7 (26.1–36.4)"], ["Male", "36 (44.4%)"], ["Female", "45 (55.6%)"], ["Total knee arthroplasty", "37 (45.7%)"], ["Total hip arthroplasty", "44 (54.3%)"], ["Primary procedure", "70 (86.4%)"], ["Revision procedure", "11 (13.6%)"], ["Elective", "67 (82.7%)"], ["Urgent/emergent", "14 (17.3%)"] ]; addTitleTable( slide, "Table 1. Baseline Characteristics", baselineData, centerX, cy, tableWidth, H * 0.31, BLUE, LIGHT_BLUE ); // ------------------------------------------------------------ // TABLE 2 // ------------------------------------------------------------ const apixabanData = [ ["Outcome", "Result"], ["Appropriate dose", "81 (100%)"], ["Dose requires optimization", "0 (0%)"], ["Appropriate duration", "10 (12.3%)"], ["Prolonged course", "40 (49.4%)"], ["Abbreviated course", "31 (38.3%)"], ["Fully appropriate regimen", "10 (12.3%)"], ["Duration requires optimization", "71 (87.7%)"], ["Median treatment duration", "30 days"], ["Treatment duration IQR", "30–35 days"] ]; addTitleTable( slide, "Table 2. Apixaban Use", apixabanData, centerX + tableWidth + tableGap, cy, tableWidth, H * 0.31, BLUE, LIGHT_BLUE ); cy += H * 0.325; // ------------------------------------------------------------ // KEY FINDING // ------------------------------------------------------------ addKeyFinding( slide, "Primary Outcome: Only 12.3% of patients received both the recommended dose and treatment duration.", centerX, cy, centerW, H * 0.055, BLUE, LIGHT_BLUE ); cy += H * 0.07; // ------------------------------------------------------------ // FIGURE 1 // ------------------------------------------------------------ const chartGap = centerW * 0.04; const chartWidth = (centerW - chartGap) / 2; drawRegimenAdherenceChart( slide, centerX, cy, chartWidth, H * 0.29, BLUE ); // ------------------------------------------------------------ // FIGURE 2 // ------------------------------------------------------------ drawDurationAppropriatenessChart( slide, centerX + chartWidth + chartGap, cy, chartWidth, H * 0.29, BLUE, APPROPRIATE, PROLONGED, ABBREVIATED ); // ============================================================ // RIGHT COLUMN // ============================================================ let ry = contentTop; // ------------------------------------------------------------ // DISCUSSION // ------------------------------------------------------------ addSectionHeader( slide, "Discussion", rightX, ry, rightW, H * 0.032, BLUE ); ry += H * 0.038; const discussionText = "• Apixaban dosing was appropriate in all included patients; however, " + "treatment duration required optimization in 87.7%.\n\n" + "• Thirty days was the most commonly prescribed treatment duration for " + "both TKA and THA.\n\n" + "• Using similar durations for both procedure types may contribute to " + "prolonged treatment following TKA and abbreviated treatment following THA.\n\n" + "• Only 2.7% of TKA patients received the recommended duration compared " + "with 20.5% of THA patients.\n\n" + "• Among TKA patients, 81.1% received prolonged treatment.\n\n" + "• Among THA patients, 56.8% received abbreviated treatment."; addBodyText( slide, discussionText, rightX, ry, rightW, H * 0.31 ); ry += H * 0.315; // ------------------------------------------------------------ // CONCLUSION // ------------------------------------------------------------ addSectionHeader( slide, "Conclusion", rightX, ry, rightW, H * 0.032, BLUE ); ry += H * 0.038; const conclusionText = "• Apixaban dose selection was appropriate in 100% of patients, but " + "only 12.3% received the recommended treatment duration.\n\n" + "• Procedure-specific treatment duration should be reinforced during " + "discharge prescribing.\n\n" + "• Recommended interventions include pharmacist review of discharge " + "prescriptions, verification of stop dates, standardized prescribing " + "workflows, targeted provider education, and EHR alerts for durations " + "that deviate from recommendations."; addBodyText( slide, conclusionText, rightX, ry, rightW, H * 0.24 ); ry += H * 0.245; // ------------------------------------------------------------ // REFERENCES // ------------------------------------------------------------ addSectionHeader( slide, "References", rightX, ry, rightW, H * 0.032, BLUE ); ry += H * 0.038; const references = "1. Lieberman JR, Heckmann N. J Am Acad Orthop Surg. 2017;25(12):789-798.\n\n" + "2. Muscatelli SR, Charters MA, Hallstrom BR. Arthroplast Today. 2021;10:105-107.\n\n" + "3. Apixaban prescribing information. Pfizer Inc; 2025.\n\n" + "4. Falck-Ytter Y, Francis CW, Johanson NA, et al. Chest. 2012;141(2 suppl):e278S-e325S.\n\n" + "5. National Institute for Health and Care Excellence. Technology appraisal guidance TA245.\n\n" + "6. Halvorsen S, Mehilli J, Cassese S, et al. Eur Heart J. 2022;43(39):3826-3924."; addText( slide, references, rightX, ry, rightW, H * 0.22, 7, DARK, false, SlidesApp.ParagraphAlignment.LEFT ); // ------------------------------------------------------------ // COMPLETE // ------------------------------------------------------------ SlidesApp.getUi().alert( "Apixaban MUE Poster Created", "The poster has been rebuilt using the three-column research poster layout.", SlidesApp.getUi().ButtonSet.OK ); } /******************************************************************** * SECTION HEADER ********************************************************************/ function addSectionHeader(slide, text, x, y, w, h, color) { const shape = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, w, h ); shape.getFill().setSolidFill(color); shape.getLine() .getLineFill() .setSolidFill(color); shape.getText().setText(text); shape.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(15) .setBold(true) .setForegroundColor("#FFFFFF"); shape.getText() .getParagraphStyle() .setParagraphAlignment( SlidesApp.ParagraphAlignment.CENTER ); shape.setContentAlignment( SlidesApp.ContentAlignment.MIDDLE ); return shape; } /******************************************************************** * BODY TEXT ********************************************************************/ function addBodyText(slide, text, x, y, w, h) { return addText( slide, text, x, y, w, h, 9, "#1F1F1F", false, SlidesApp.ParagraphAlignment.LEFT ); } /******************************************************************** * GENERIC TEXT BOX ********************************************************************/ function addText( slide, text, x, y, w, h, fontSize, color, bold, alignment ) { const box = slide.insertTextBox( text, x, y, w, h ); box.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(fontSize) .setForegroundColor(color) .setBold(bold); box.getText() .getParagraphStyle() .setParagraphAlignment(alignment); return box; } /******************************************************************** * OUTLINED BOX ********************************************************************/ function addOutlinedBox(slide, x, y, w, h, color) { const box = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, w, h ); box.getFill().setTransparent(); box.getLine() .getLineFill() .setSolidFill(color); return box; } /******************************************************************** * METHODS TABLE ********************************************************************/ function addTwoColumnTable( slide, data, x, y, w, h, headerColor, lightColor ) { const table = slide.insertTable( data.length, 2, x, y, w, h ); for (let r = 0; r < data.length; r++) { for (let c = 0; c < 2; c++) { const cell = table.getCell(r, c); cell.getText().setText( data[r][c] ); cell.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(7) .setForegroundColor("#1F1F1F"); cell.setContentAlignment( SlidesApp.ContentAlignment.MIDDLE ); if (c === 0) { cell.getFill() .setSolidFill(lightColor); cell.getText() .getTextStyle() .setBold(true) .setForegroundColor(headerColor); } } } return table; } /******************************************************************** * TITLED RESULT TABLE ********************************************************************/ function addTitleTable( slide, title, data, x, y, w, h, headerColor, lightColor ) { const titleH = h * 0.10; const titleShape = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, w, titleH ); titleShape.getFill() .setSolidFill(headerColor); titleShape.getLine() .getLineFill() .setSolidFill(headerColor); titleShape.getText() .setText(title); titleShape.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(9) .setBold(true) .setForegroundColor("#FFFFFF"); titleShape.getText() .getParagraphStyle() .setParagraphAlignment( SlidesApp.ParagraphAlignment.CENTER ); const table = slide.insertTable( data.length, data[0].length, x, y + titleH, w, h - titleH ); for (let r = 0; r < data.length; r++) { for (let c = 0; c < data[r].length; c++) { const cell = table.getCell(r, c); cell.getText() .setText(data[r][c]); cell.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(6.5) .setForegroundColor("#1F1F1F"); cell.setContentAlignment( SlidesApp.ContentAlignment.MIDDLE ); if (r === 0) { cell.getFill() .setSolidFill(lightColor); cell.getText() .getTextStyle() .setBold(true); } } } return table; } /******************************************************************** * KEY FINDING BOX ********************************************************************/ function addKeyFinding( slide, text, x, y, w, h, blue, lightBlue ) { const shape = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, w, h ); shape.getFill() .setSolidFill(lightBlue); shape.getLine() .getLineFill() .setSolidFill(blue); shape.getText() .setText(text); shape.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(10) .setBold(true) .setForegroundColor(blue); shape.getText() .getParagraphStyle() .setParagraphAlignment( SlidesApp.ParagraphAlignment.CENTER ); shape.setContentAlignment( SlidesApp.ContentAlignment.MIDDLE ); } /******************************************************************** * FIGURE 1 * APIXABAN REGIMEN ADHERENCE ********************************************************************/ function drawRegimenAdherenceChart( slide, x, y, w, h, blue ) { addText( slide, "Figure 1. Apixaban Regimen Adherence", x, y, w, h * 0.11, 9, "#1F1F1F", true, SlidesApp.ParagraphAlignment.CENTER ); const chartTop = y + h * 0.17; const chartBottom = y + h * 0.83; const chartH = chartBottom - chartTop; const baseline = chartBottom; // Y axis slide.insertLine( SlidesApp.LineCategory.STRAIGHT, x + w * 0.15, chartTop, x + w * 0.15, baseline ); // X axis slide.insertLine( SlidesApp.LineCategory.STRAIGHT, x + w * 0.15, baseline, x + w * 0.92, baseline ); const barW = w * 0.18; const bar1X = x + w * 0.30; const bar2X = x + w * 0.65; const bar1H = chartH; const bar2H = chartH * 0.123; addBar( slide, bar1X, baseline - bar1H, barW, bar1H, blue, "100%" ); addBar( slide, bar2X, baseline - bar2H, barW, bar2H, blue, "12.3%" ); addText( slide, "Correct\nDose", bar1X - w * 0.03, baseline + h * 0.02, barW + w * 0.06, h * 0.12, 7, "#1F1F1F", true, SlidesApp.ParagraphAlignment.CENTER ); addText( slide, "Correct\nDuration", bar2X - w * 0.03, baseline + h * 0.02, barW + w * 0.06, h * 0.12, 7, "#1F1F1F", true, SlidesApp.ParagraphAlignment.CENTER ); } /******************************************************************** * SIMPLE BAR ********************************************************************/ function addBar( slide, x, y, w, h, color, label ) { const bar = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, w, h ); bar.getFill() .setSolidFill(color); bar.getLine() .getLineFill() .setSolidFill(color); addText( slide, label, x, Math.max(y - 15, 0), w, 14, 7, color, true, SlidesApp.ParagraphAlignment.CENTER ); } /******************************************************************** * FIGURE 2 * PROCEDURE-SPECIFIC DURATION APPROPRIATENESS ********************************************************************/ function drawDurationAppropriatenessChart( slide, x, y, w, h, blue, appropriateColor, prolongedColor, abbreviatedColor ) { addText( slide, "Figure 2. Duration Appropriateness by Procedure Type", x, y, w, h * 0.11, 9, "#1F1F1F", true, SlidesApp.ParagraphAlignment.CENTER ); const labelW = w * 0.18; const barX = x + labelW; const barW = w * 0.78; const barH = h * 0.13; // ------------------------------------------------------------ // HIP // ------------------------------------------------------------ const hipY = y + h * 0.30; addText( slide, "THA", x, hipY, labelW * 0.9, barH, 8, "#1F1F1F", true, SlidesApp.ParagraphAlignment.CENTER ); drawStackedBar( slide, barX, hipY, barW, barH, [ [20.5, appropriateColor], [22.7, prolongedColor], [56.8, abbreviatedColor] ] ); // ------------------------------------------------------------ // KNEE // ------------------------------------------------------------ const kneeY = y + h * 0.52; addText( slide, "TKA", x, kneeY, labelW * 0.9, barH, 8, "#1F1F1F", true, SlidesApp.ParagraphAlignment.CENTER ); drawStackedBar( slide, barX, kneeY, barW, barH, [ [2.7, appropriateColor], [81.1, prolongedColor], [16.2, abbreviatedColor] ] ); // ------------------------------------------------------------ // LEGEND // ------------------------------------------------------------ const legendY = y + h * 0.75; addLegendItem( slide, x + w * 0.02, legendY, appropriateColor, "Appropriate" ); addLegendItem( slide, x + w * 0.35, legendY, prolongedColor, "Prolonged" ); addLegendItem( slide, x + w * 0.67, legendY, abbreviatedColor, "Abbreviated" ); } /******************************************************************** * STACKED BAR ********************************************************************/ function drawStackedBar( slide, x, y, w, h, values ) { let currentX = x; for (let i = 0; i < values.length; i++) { const percent = values[i][0]; const color = values[i][1]; const segmentW = w * (percent / 100); const segment = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, currentX, y, segmentW, h ); segment.getFill() .setSolidFill(color); segment.getLine() .getLineFill() .setSolidFill(color); if (segmentW > 18) { segment.getText() .setText( percent.toFixed(1) + "%" ); segment.getText() .getTextStyle() .setFontFamily("Arial") .setFontSize(6) .setBold(true) .setForegroundColor("#FFFFFF"); segment.getText() .getParagraphStyle() .setParagraphAlignment( SlidesApp.ParagraphAlignment.CENTER ); segment.setContentAlignment( SlidesApp.ContentAlignment.MIDDLE ); } currentX += segmentW; } } /******************************************************************** * LEGEND ITEM ********************************************************************/ function addLegendItem( slide, x, y, color, label ) { const square = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, 8, 8 ); square.getFill() .setSolidFill(color); square.getLine() .getLineFill() .setSolidFill(color); addText( slide, label, x + 10, y - 2, 60, 12, 6, "#1F1F1F", false, SlidesApp.ParagraphAlignment.LEFT ); } /******************************************************************** * MENU ********************************************************************/ function onOpen() { SlidesApp.getUi() .createMenu("Apixaban MUE") .addItem( "Build Poster", "buildApixabanPoster" ) .addToUi(); }
