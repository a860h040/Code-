right.setContentAlignment( SlidesApp.ContentAlignment.MIDDLE ); } } /********************************************************************** * RESULTS TABLE **********************************************************************/ function addResultsTable( slide, title, data, x, y, w, h, C ) { const titleH = h * 0.095; } } } } } /********************************************************************** * PRIMARY OUTCOME HIGHLIGHT **********************************************************************/ function addHighlightBox( slide, text, x, y, w, h, C ) { const box = slide.insertShape( SlidesApp.ShapeType.RECTANGLE, x, y, w, h ); } } /********************************************************************** * FIGURE 2 **********************************************************************/ function drawDurationChart( slide, x, y, w, h, C ) { addText( slide, "Figure 2: Duration Appropriateness by Procedure Type", x, y, w, h * 0.10, 14, C.black, true, SlidesApp.ParagraphAlignment.CENTER ); } /********************************************************************** * STACKED BAR **********************************************************************/ function stackedBar( slide, x, y, w, h, segments ) { let currentX = x; square.getFill() .setSolidFill(color); try { square.getLine() .getFill() .setSolidFill(color); } catch (e) {} addText( slide, label, x + 20, y - 4, 120, 22, 9, C.black, false, SlidesApp.ParagraphAlignment.START );
