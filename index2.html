const SETTINGS = { SHEET_NAME: "Calendar Feed", // Leave blank to use your primary Google Calendar. // For another calendar, enter its Calendar ID. CALENDAR_ID: "", // Events included in the feed. DAYS_IN_THE_PAST: 30, DAYS_IN_THE_FUTURE: 365 }; /** * Adds a Calendar Feed menu when the spreadsheet opens. */ function onOpen() { SpreadsheetApp.getUi() .createMenu("Calendar Feed") .addItem("Sync calendar now", "syncCalendarToSheet") .addSeparator() .addItem("Install automatic sync", "installAutomaticSync") .addItem("Remove automatic sync", "removeAutomaticSync") .addToUi(); } /** * Reads calendar events and writes them into the Google Sheet. */ function syncCalendarToSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); let sheet = spreadsheet.getSheetByName(SETTINGS.SHEET_NAME); if (!sheet) { sheet = spreadsheet.insertSheet(SETTINGS.SHEET_NAME); } const calendar = SETTINGS.CALENDAR_ID.trim() ? CalendarApp.getCalendarById(SETTINGS.CALENDAR_ID.trim()) : CalendarApp.getDefaultCalendar(); if (!calendar) { throw new Error( "Calendar not found. Check the Calendar ID and confirm that you have access." ); } const startDate = new Date(); startDate.setDate(startDate.getDate() - SETTINGS.DAYS_IN_THE_PAST); startDate.setHours(0, 0, 0, 0); const endDate = new Date(); endDate.setDate(endDate.getDate() + SETTINGS.DAYS_IN_THE_FUTURE); endDate.setHours(23, 59, 59, 999); const events = calendar .getEvents(startDate, endDate) .sort((a, b) => a.getStartTime() - b.getStartTime()); const headers = [ "Event Key", "Event ID", "Calendar", "Event Title", "Start Date and Time", "End Date and Time", "All-Day Event", "Location", "Description", "Created By", "Guests", "My Status", "Recurring Event", "Last Updated" ]; const rows = events.map(event => { const startTime = event.getStartTime(); const eventKey = event.getId() + "|" + startTime.toISOString(); const guests = event .getGuestList() .map(guest => guest.getEmail()) .filter(email => email) .join(", "); return [ eventKey, event.getId(), calendar.getName(), event.getTitle(), startTime, event.getEndTime(), event.isAllDayEvent() ? "Yes" : "No", event.getLocation() || "", event.getDescription() || "", event.getCreators().join(", "), guests, event.getMyStatus().toString(), event.isRecurringEvent() ? "Yes" : "No", event.getLastUpdated() ]; }); // Remove the existing filter before replacing the data. const existingFilter = sheet.getFilter(); if (existingFilter) { existingFilter.remove(); } sheet.clearContents(); sheet .getRange(1, 1, 1, headers.length) .setValues([headers]) .setFontWeight("bold") .setBackground("#1a73e8") .setFontColor("#ffffff"); if (rows.length > 0) { sheet .getRange(2, 1, rows.length, headers.length) .setValues(rows); // Start and end dates. sheet .getRange(2, 5, rows.length, 2) .setNumberFormat("mm/dd/yyyy hh:mm AM/PM"); // Last updated date. sheet .getRange(2, 14, rows.length, 1) .setNumberFormat("mm/dd/yyyy hh:mm AM/PM"); sheet .getRange(1, 1, rows.length + 1, headers.length) .createFilter(); } sheet.setFrozenRows(1); sheet.setFrozenColumns(1); sheet.setColumnWidth(1, 290); sheet.setColumnWidth(2, 250); sheet.setColumnWidth(3, 170); sheet.setColumnWidth(4, 250); sheet.setColumnWidth(5, 170); sheet.setColumnWidth(6, 170); sheet.setColumnWidth(7, 110); sheet.setColumnWidth(8, 220); sheet.setColumnWidth(9, 350); sheet.setColumnWidth(10, 220); sheet.setColumnWidth(11, 300); sheet.setColumnWidth(12, 120); sheet.setColumnWidth(13, 120); sheet.setColumnWidth(14, 170); sheet .getDataRange() .setVerticalAlignment("middle"); sheet .getRange(2, 8, Math.max(rows.length, 1), 4) .setWrap(true); spreadsheet.toast( rows.length + " calendar events imported.", "Calendar Feed", 5 ); } /** * Creates a trigger that refreshes the calendar feed every 15 minutes. */ function installAutomaticSync() { removeAutomaticSync(false); ScriptApp.newTrigger("syncCalendarToSheet") .timeBased() .everyMinutes(15) .create(); syncCalendarToSheet(); SpreadsheetApp.getUi().alert( "Automatic calendar synchronization has been installed. " + "The sheet will refresh approximately every 15 minutes." ); } /** * Removes the automatic Calendar Feed triggers. */ function removeAutomaticSync(showMessage = true) { const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(trigger => { if (trigger.getHandlerFunction() === "syncCalendarToSheet") { ScriptApp.deleteTrigger(trigger); } }); if (showMessage) { SpreadsheetApp.getUi().alert( "Automatic calendar synchronization has been removed." ); } }
