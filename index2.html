/* ========================================================================== * ORTHO → SHEET1 AUTOFILL * Paste this entire section at the BOTTOM of Code.gs * ========================================================================== */ function runOrthoAutofill() { const result = orthoAutofillAllRows_(); const message = 'Ortho autofill completed.\n\n' + 'Sheet1 rows checked: ' + result.rowsChecked + '\n' + 'FINs matched: ' + result.finsMatched + '\n' + 'Rows updated: ' + result.rowsUpdated + '\n' + 'Cells filled: ' + result.cellsFilled + '\n' + 'FINs not found: ' + result.notFound.length + '\n' + 'Duplicate FINs skipped: ' + result.duplicates.length + '\n' + 'Blank FIN rows skipped: ' + result.blankFinRows.length; console.log(message); console.log(JSON.stringify(result, null, 2)); try { SpreadsheetApp.getActiveSpreadsheet().toast( message, 'Ortho Autofill', 10 ); } catch (error) { console.log('Toast could not be displayed: ' + error.message); } return result; } function orthoAutofillAllRows_() { const spreadsheet = orthoGetSpreadsheet_(); const sourceSheet = spreadsheet.getSheetByName('Ortho'); const destinationSheet = spreadsheet.getSheetByName('Sheet1'); if (!sourceSheet) { throw new Error( 'The source sheet named "Ortho" was not found. ' + 'Confirm that the tab name is exactly Ortho.' ); } if (!destinationSheet) { throw new Error( 'The destination sheet named "Sheet1" was not found. ' + 'Confirm that the tab name is exactly Sheet1.' ); } if (sourceSheet.getLastRow() < 2) { throw new Error('The Ortho sheet does not contain any patient data.'); } if (destinationSheet.getLastRow() < 2) { throw new Error( 'Sheet1 does not contain any patient rows. ' + 'Enter FIN numbers in Sheet1 before running the autofill.' ); } const sourceLastColumn = sourceSheet.getLastColumn(); const destinationLastColumn = destinationSheet.getLastColumn(); const sourceHeaders = sourceSheet .getRange(1, 1, 1, sourceLastColumn) .getDisplayValues()[0]; const destinationHeaders = destinationSheet .getRange(1, 1, 1, destinationLastColumn) .getDisplayValues()[0]; const sourceHeaderMap = orthoCreateHeaderMap_(sourceHeaders, 'Ortho'); const destinationHeaderMap = orthoCreateHeaderMap_( destinationHeaders, 'Sheet1' ); const requiredOrthoHeaders = [ 'PATIENT', 'FIN', 'ORD_ID', 'DOB', 'AGE', 'GENDER', 'ORIG_ORDER_DT_TM', 'O_DEPT_STATUS_DISP', 'O_CATALOG_DISP', 'ORDERED_AS_MNEMONIC', 'CLINICAL_DISPLAY_LINE', 'PROVIDER', 'ORDER_DC', 'ORDER_DC_TYPE', 'E_ENCNTR_TYPE_DISP', 'ENC_ARRIVE_DT_TM', 'ENC_DISCH_DT_TM', 'E_LOC_FACILITY_DISP', 'E_LOC_NURSE_UNIT_DISP', 'ORTHO Hip Procedures Post-Op', 'PERSON_ID', 'ENCNTR_ID' ]; const missingOrthoHeaders = requiredOrthoHeaders.filter(function (header) { return sourceHeaderMap[orthoNormalizeHeader_(header)] === undefined; }); if (missingOrthoHeaders.length > 0) { throw new Error( 'The following required headers are missing from Ortho:\n' + missingOrthoHeaders.join(', ') ); } const sourceFinColumn = sourceHeaderMap[orthoNormalizeHeader_('FIN')]; const destinationFinColumn = destinationHeaderMap[orthoNormalizeHeader_('FIN')]; if (destinationFinColumn === undefined) { throw new Error( 'The FIN header was not found in row 1 of Sheet1.' ); } const sourceRowCount = sourceSheet.getLastRow() - 1; const destinationRowCount = destinationSheet.getLastRow() - 1; const sourceValues = sourceSheet .getRange(2, 1, sourceRowCount, sourceLastColumn) .getValues(); const sourceDisplayValues = sourceSheet .getRange(2, 1, sourceRowCount, sourceLastColumn) .getDisplayValues(); const destinationRange = destinationSheet.getRange( 2, 1, destinationRowCount, destinationLastColumn ); const destinationValues = destinationRange.getValues(); const destinationDisplayValues = destinationRange.getDisplayValues(); const destinationFormulas = destinationRange.getFormulas(); const sourceRecordsByFin = {}; const duplicateFinSet = {}; const blankSourceFinRows = []; sourceValues.forEach(function (row, rowIndex) { const displayedFin = sourceDisplayValues[rowIndex][sourceFinColumn]; const fin = orthoNormalizeFin_(displayedFin); if (!fin) { blankSourceFinRows.push(rowIndex + 2); return; } if (sourceRecordsByFin[fin]) { duplicateFinSet[fin] = true; return; } sourceRecordsByFin[fin] = { values: row, displayValues: sourceDisplayValues[rowIndex], sourceRow: rowIndex + 2 }; }); Object.keys(duplicateFinSet).forEach(function (fin) { delete sourceRecordsByFin[fin]; }); const assignments = orthoBuildAssignments_( sourceHeaderMap, destinationHeaderMap ); if (assignments.length === 0) { throw new Error( 'No matching destination columns were found between Ortho and Sheet1.' ); } let rowsUpdated = 0; let cellsFilled = 0; let finsMatched = 0; const notFound = []; const duplicates = []; const blankFinRows = []; destinationValues.forEach(function (destinationRow, rowIndex) { const displayedDestinationFin = destinationDisplayValues[rowIndex][destinationFinColumn]; const fin = orthoNormalizeFin_(displayedDestinationFin); const sheetRowNumber = rowIndex + 2; if (!fin) { blankFinRows.push(sheetRowNumber); return; } if (duplicateFinSet[fin]) { duplicates.push({ row: sheetRowNumber, fin: fin }); return; } const sourceRecord = sourceRecordsByFin[fin]; if (!sourceRecord) { notFound.push({ row: sheetRowNumber, fin: fin }); return; } finsMatched++; let rowChanged = false; assignments.forEach(function (assignment) { const destinationColumn = assignment.destinationColumn; const existingValue = destinationRow[destinationColumn]; const existingDisplayValue = destinationDisplayValues[rowIndex][destinationColumn]; const existingFormula = destinationFormulas[rowIndex][destinationColumn]; // Never overwrite formulas. if (existingFormula) { return; } // Fill only truly empty cells. if ( existingValue !== '' && existingValue !== null && existingValue !== undefined ) { return; } if (String(existingDisplayValue || '').trim() !== '') { return; } let sourceValue; if (assignment.type === 'lengthOfStay') { sourceValue = orthoCalculateLengthOfStay_( sourceRecord.values, sourceHeaderMap ); } else { sourceValue = sourceRecord.values[assignment.sourceColumn]; } if ( sourceValue === '' || sourceValue === null || sourceValue === undefined ) { return; } destinationRow[destinationColumn] = sourceValue; rowChanged = true; cellsFilled++; }); if (rowChanged) { rowsUpdated++; } }); destinationRange.setValues(destinationValues); SpreadsheetApp.flush(); return { success: true, rowsChecked: destinationRowCount, finsMatched: finsMatched, rowsUpdated: rowsUpdated, cellsFilled: cellsFilled, blankFinRows: blankFinRows, blankSourceFinRows: blankSourceFinRows, notFound: notFound, duplicates: duplicates, mappedColumns: assignments.map(function (assignment) { return assignment.sourceHeader + ' → ' + assignment.destinationHeader; }) }; } function orthoBuildAssignments_( sourceHeaderMap, destinationHeaderMap ) { const assignments = []; const usedDestinationColumns = {}; // Copy columns having the same header name. Object.keys(destinationHeaderMap).forEach(function ( normalizedDestinationHeader ) { if (normalizedDestinationHeader === orthoNormalizeHeader_('FIN')) { return; } if ( sourceHeaderMap[normalizedDestinationHeader] !== undefined ) { const sourceColumn = sourceHeaderMap[normalizedDestinationHeader]; const destinationColumn = destinationHeaderMap[normalizedDestinationHeader]; assignments.push({ type: 'direct', sourceColumn: sourceColumn, destinationColumn: destinationColumn, sourceHeader: normalizedDestinationHeader, destinationHeader: normalizedDestinationHeader }); usedDestinationColumns[destinationColumn] = true; } }); const aliases = [ { source: 'AGE', destination: 'Age' }, { source: 'GENDER', destination: 'Gender' }, { source: 'ORTHO Hip Procedures Post-Op', destination: 'Type of surgery' }, { source: 'ORIG_ORDER_DT_TM', destination: 'Date and time of surgery' }, { source: 'PROVIDER', destination: 'Surgeon' } ]; aliases.forEach(function (alias) { const sourceKey = orthoNormalizeHeader_(alias.source); const destinationKey = orthoNormalizeHeader_(alias.destination); const sourceColumn = sourceHeaderMap[sourceKey]; const destinationColumn = destinationHeaderMap[destinationKey]; if ( sourceColumn === undefined || destinationColumn === undefined || usedDestinationColumns[destinationColumn] ) { return; } assignments.push({ type: 'direct', sourceColumn: sourceColumn, destinationColumn: destinationColumn, sourceHeader: alias.source, destinationHeader: alias.destination }); usedDestinationColumns[destinationColumn] = true; }); const lengthOfStayColumn = destinationHeaderMap[ orthoNormalizeHeader_('Length of stay') ]; if ( lengthOfStayColumn !== undefined && !usedDestinationColumns[lengthOfStayColumn] ) { assignments.push({ type: 'lengthOfStay', sourceColumn: null, destinationColumn: lengthOfStayColumn, sourceHeader: 'ENC_ARRIVE_DT_TM and ENC_DISCH_DT_TM', destinationHeader: 'Length of stay' }); } return assignments; } function orthoCalculateLengthOfStay_( sourceRow, sourceHeaderMap ) { const arrivalColumn = sourceHeaderMap[ orthoNormalizeHeader_('ENC_ARRIVE_DT_TM') ]; const dischargeColumn = sourceHeaderMap[ orthoNormalizeHeader_('ENC_DISCH_DT_TM') ]; if ( arrivalColumn === undefined || dischargeColumn === undefined ) { return ''; } const arrival = orthoConvertToDate_( sourceRow[arrivalColumn] ); const discharge = orthoConvertToDate_( sourceRow[dischargeColumn] ); if (!arrival || !discharge) { return ''; } const milliseconds = discharge.getTime() - arrival.getTime(); if (milliseconds < 0) { return ''; } const days = milliseconds / (1000 * 60 * 60 * 24); return Math.round(days * 100) / 100; } function orthoConvertToDate_(value) { if (value instanceof Date && !isNaN(value.getTime())) { return value; } if (!value) { return null; } const converted = new Date(value); if (isNaN(converted.getTime())) { return null; } return converted; } function orthoCreateHeaderMap_(headers, sheetName) { const map = {}; const duplicates = []; headers.forEach(function (header, columnIndex) { const normalizedHeader = orthoNormalizeHeader_(header); if (!normalizedHeader) { return; } if (map[normalizedHeader] !== undefined) { duplicates.push(String(header)); return; } map[normalizedHeader] = columnIndex; }); if (duplicates.length > 0) { throw new Error( 'Duplicate column headers were found in ' + sheetName + ': ' + duplicates.join(', ') ); } return map; } function orthoNormalizeHeader_(value) { return String(value || '') .trim() .replace(/\s+/g, ' ') .toUpperCase(); } function orthoNormalizeFin_(value) { return String(value || '') .trim() .replace(/\s+/g, ''); } function orthoGetSpreadsheet_() { const activeSpreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (activeSpreadsheet) { return activeSpreadsheet; } if ( typeof CONFIG !== 'undefined' && CONFIG.SPREADSHEET_ID ) { return SpreadsheetApp.openById( CONFIG.SPREADSHEET_ID ); } throw new Error( 'The Apps Script project is not connected to a Google Sheet. ' + 'Open the MUE spreadsheet and select Extensions → Apps Script, ' + 'or enter the spreadsheet ID in CONFIG.SPREADSHEET_ID.' ); }
