/********************************************************************** * * UPDATED FOUR-SECTION PIE CHART * * FIXES BLANK PIE CHART ISSUE * * FOUR SECTIONS: * * 1. Total Hip – Appropriate Duration * 2. Total Hip – Inappropriate Duration * 3. Total Knee – Appropriate Duration * 4. Total Knee – Inappropriate Duration * * The pie slices show percentages. * The legend shows patient counts (n). * * Together the four sections = 100%. * **********************************************************************/ function buildOverallPieSheet_(ctx) { const sh = resetSheet_( ctx.ss, MUE.SHEETS.OVERALL_PIE ); /******************************************************************** * TITLE ********************************************************************/ addSheetTitle_( sh, "Apixaban Duration Appropriateness by Surgery Type" ); /******************************************************************** * DESCRIPTION ********************************************************************/ sh .getRange("A2:H2") .merge() .setValue( "Four-section pie chart showing total hip and total knee " + "arthroplasty patients by duration appropriateness. " + "Each percentage uses all included hip/knee patients as the denominator." ) .setWrap(true); sh .getRange("A3:H3") .merge() .setValue( "Inappropriate Duration includes Too Long, Too Short, and " + "Unable to Determine so that every included hip/knee patient " + "is represented and the four sections equal 100%." ) .setWrap(true) .setFontStyle("italic"); const a = ctx.auditName; const r = ctx.auditLastRow; /******************************************************************** * * MAIN SUMMARY TABLE * ********************************************************************/ sh .getRange("A5:C10") .setValues([ [ "Category", "Patient Count (n)", "% of Included Patients" ], [ "Total Hip – Appropriate Duration", "", "" ], [ "Total Hip – Inappropriate Duration", "", "" ], [ "Total Knee – Appropriate Duration", "", "" ], [ "Total Knee – Inappropriate Duration", "", "" ], [ "TOTAL", "", "" ] ]); styleHeader_( sh.getRange("A5:C5") ); /******************************************************************** * HIP APPROPRIATE ********************************************************************/ sh .getRange("B6") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Hip",` + `${a}!$I$5:$I$${r},"Appropriate"` + `)` ); /******************************************************************** * HIP INAPPROPRIATE * * Total Hip - Appropriate Hip ********************************************************************/ sh .getRange("B7") .setFormula( `=COUNTIF(` + `${a}!$B$5:$B$${r},"Hip"` + `)-B6` ); /******************************************************************** * KNEE APPROPRIATE ********************************************************************/ sh .getRange("B8") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Knee",` + `${a}!$I$5:$I$${r},"Appropriate"` + `)` ); /******************************************************************** * KNEE INAPPROPRIATE * * Total Knee - Appropriate Knee ********************************************************************/ sh .getRange("B9") .setFormula( `=COUNTIF(` + `${a}!$B$5:$B$${r},"Knee"` + `)-B8` ); /******************************************************************** * TOTAL INCLUDED PATIENTS ********************************************************************/ sh .getRange("B10") .setFormula( "=SUM(B6:B9)" ); /******************************************************************** * PERCENTAGES * * All use same denominator. ********************************************************************/ sh .getRange("C6") .setFormula( "=IFERROR(B6/$B$10,0)" ); sh .getRange("C7") .setFormula( "=IFERROR(B7/$B$10,0)" ); sh .getRange("C8") .setFormula( "=IFERROR(B8/$B$10,0)" ); sh .getRange("C9") .setFormula( "=IFERROR(B9/$B$10,0)" ); sh .getRange("C10") .setFormula( "=SUM(C6:C9)" ); sh .getRange("C6:C10") .setNumberFormat( "0.0%" ); sh .getRange("A5:C10") .setBorder( true, true, true, true, true, true ); sh .getRange("A10:C10") .setFontWeight( "bold" ); /******************************************************************** * * VALIDATION TABLE * ********************************************************************/ sh .getRange("E5:F9") .setValues([ [ "Validation", "Result" ], [ "Included Hip Patients", "" ], [ "Included Knee Patients", "" ], [ "Total Included Patients", "" ], [ "Pie Total %", "" ] ]); styleHeader_( sh.getRange("E5:F5") ); sh .getRange("F6") .setFormula( `=COUNTIF(` + `${a}!$B$5:$B$${r},"Hip"` + `)` ); sh .getRange("F7") .setFormula( `=COUNTIF(` + `${a}!$B$5:$B$${r},"Knee"` + `)` ); sh .getRange("F8") .setFormula( "=F6+F7" ); sh .getRange("F9") .setFormula( "=SUM(C6:C9)" ); sh .getRange("F9") .setNumberFormat( "0.0%" ); sh .getRange("E5:F9") .setBorder( true, true, true, true, true, true ); /******************************************************************** * * PIE CHART SOURCE TABLE * * IMPORTANT FIX: * * Keep labels shorter. * * Counts are included in the legend label: * * Total Hip – Appropriate (n=9) * * The actual pie slice displays: * * 11.1% * ********************************************************************/ sh .getRange("H5:I9") .setValues([ [ "Category", "Patients" ], [ "", "" ], [ "", "" ], [ "", "" ], [ "", "" ] ]); styleHeader_( sh.getRange("H5:I5") ); /******************************************************************** * HIP APPROPRIATE ********************************************************************/ sh .getRange("H6") .setFormula( `="Hip – Appropriate (n="&B6&")"` ); sh .getRange("I6") .setFormula( "=B6" ); /******************************************************************** * HIP INAPPROPRIATE ********************************************************************/ sh .getRange("H7") .setFormula( `="Hip – Inappropriate (n="&B7&")"` ); sh .getRange("I7") .setFormula( "=B7" ); /******************************************************************** * KNEE APPROPRIATE ********************************************************************/ sh .getRange("H8") .setFormula( `="Knee – Appropriate (n="&B8&")"` ); sh .getRange("I8") .setFormula( "=B8" ); /******************************************************************** * KNEE INAPPROPRIATE ********************************************************************/ sh .getRange("H9") .setFormula( `="Knee – Inappropriate (n="&B9&")"` ); sh .getRange("I9") .setFormula( "=B9" ); sh .getRange("H5:I9") .setBorder( true, true, true, true, true, true ); /******************************************************************** * * INAPPROPRIATE DURATION BREAKDOWN * ********************************************************************/ sh .getRange("E12:H16") .setValues([ [ "Procedure", "Too Long", "Too Short", "Unable to Determine" ], [ "Hip", "", "", "" ], [ "Knee", "", "", "" ], [ "", "", "", "" ], [ "Note", "Too Long + Too Short + Unable to Determine are grouped as Inappropriate Duration.", "", "" ] ]); styleHeader_( sh.getRange("E12:H12") ); /******************************************************************** * HIP BREAKDOWN ********************************************************************/ sh .getRange("F13") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Hip",` + `${a}!$I$5:$I$${r},"Too long"` + `)` ); sh .getRange("G13") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Hip",` + `${a}!$I$5:$I$${r},"Too short"` + `)` ); sh .getRange("H13") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Hip",` + `${a}!$I$5:$I$${r},"Unable to determine"` + `)` ); /******************************************************************** * KNEE BREAKDOWN ********************************************************************/ sh .getRange("F14") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Knee",` + `${a}!$I$5:$I$${r},"Too long"` + `)` ); sh .getRange("G14") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Knee",` + `${a}!$I$5:$I$${r},"Too short"` + `)` ); sh .getRange("H14") .setFormula( `=COUNTIFS(` + `${a}!$B$5:$B$${r},"Knee",` + `${a}!$I$5:$I$${r},"Unable to determine"` + `)` ); sh .getRange("E12:H16") .setBorder( true, true, true, true, true, true ) .setWrap(true); /******************************************************************** * COLUMN WIDTHS ********************************************************************/ sh.setColumnWidth( 1, 300 ); sh.setColumnWidth( 2, 130 ); sh.setColumnWidth( 3, 160 ); sh.setColumnWidth( 5, 180 ); sh.setColumnWidth( 6, 140 ); sh.setColumnWidth( 8, 280 ); sh.setColumnWidth( 9, 100 ); /******************************************************************** * * CRITICAL FIX * * Force Google Sheets to finish calculating B6:B10 and H6:I9 * BEFORE the chart is created. * ********************************************************************/ SpreadsheetApp.flush(); Utilities.sleep( 1500 ); SpreadsheetApp.flush(); /******************************************************************** * REMOVE ANY OLD CHARTS ********************************************************************/ sh .getCharts() .forEach( function(chart) { sh.removeChart( chart ); } ); /******************************************************************** * * CREATE PIE CHART * * FIX: * * pieSliceText = percentage * * instead of: * * pieSliceText = label * ********************************************************************/ const chart = sh .newChart() .asPieChart() .addRange( sh.getRange( "H5:I9" ) ) .setNumHeaders( 1 ) .setPosition( 18, 1, 0, 0 ) .setOption( "title", "Apixaban Duration Appropriateness by Surgery Type" ) .setOption( "legend", { position: "right", textStyle: { fontSize: 11 } } ) /**************************************************************** * SHOW % DIRECTLY ON EACH PIE SLICE ****************************************************************/ .setOption( "pieSliceText", "percentage" ) .setOption( "pieSliceTextStyle", { fontSize: 12 } ) /**************************************************************** * SHOW EVEN VERY SMALL SLICE * * Important because Knee Appropriate may only be ~1%. ****************************************************************/ .setOption( "sliceVisibilityThreshold", 0 ) .setOption( "width", 950 ) .setOption( "height", 600 ) .build(); sh.insertChart( chart ); /******************************************************************** * FINAL FLUSH ********************************************************************/ SpreadsheetApp.flush(); Utilities.sleep( 500 ); SpreadsheetApp.flush(); finishGraphSheet_( sh ); }
