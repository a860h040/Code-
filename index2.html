/********************************************************************** * * RESOLVE FACILITY CANDIDATES * **********************************************************************/ function resolveFacilityCandidates_( candidates, sourceDateTime ) { if ( !candidates || candidates.length === 0 ) { return { facility: "Unknown / unmatched", rawFacility: "", status: "No match", rows: "" }; } /******************************************************************** * UNIQUE FACILITIES ********************************************************************/ const uniqueFacilities = Array.from( new Set( candidates .map( c => c.facility ) .filter( Boolean ) ) ); /******************************************************************** * ONE FACILITY * * Multiple Ortho rows may represent duplicate/order records. ********************************************************************/ if ( uniqueFacilities.length === 1 ) { return { facility: uniqueFacilities[0], rawFacility: candidates[0] .rawFacility, status: candidates.length === 1 ? "Matched" : "Matched - duplicate Ortho rows", rows: candidates .map( c => c.orthoRow ) .join(", ") }; } /******************************************************************** * IF MULTIPLE FACILITIES: * TRY EXACT / CLOSEST DATE-TIME ********************************************************************/ if ( sourceDateTime !== null ) { const sorted = candidates .filter( c => c.surgeryDateTime !== null ) .map( c => { return { candidate: c, distance: Math.abs( c.surgeryDateTime - sourceDateTime ) }; } ) .sort( (a, b) => a.distance - b.distance ); if ( sorted.length > 0 ) { const best = sorted[0]; const second = sorted.length > 1 ? sorted[1] : null; /**************************************************************** * Require a unique closest time. ****************************************************************/ if ( !second || best.distance < second.distance ) { return { facility: best.candidate .facility, rawFacility: best.candidate .rawFacility, status: "Matched - closest surgery time", rows: best.candidate .orthoRow .toString() }; } } } /******************************************************************** * AMBIGUOUS ********************************************************************/ return { facility: "Unknown / ambiguous", rawFacility: candidates .map( c => c.rawFacility ) .join(" | "), status: "Ambiguous", rows: candidates .map( c => c.orthoRow ) .join(", ") };
